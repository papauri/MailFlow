   547	function formatSize(bytes: number) {
   548	  if (!bytes || bytes === 0) return '0 B';
   549	  const k = 1024;
   550	  const sizes = ['B', 'KB', 'MB', 'GB'];
   551	  const i = Math.floor(Math.log(bytes) / Math.log(k));
   552	  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
   553	}
